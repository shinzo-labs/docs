---
title: "Create Token"
api: "POST https://api.app.shinzo.ai/v1/tokens"
description: "Generate a new authentication token"
---

Create a new authentication token with specified permissions. The token value is only returned once during creation.

<Warning>
The token value is only displayed once. Store it securely as it cannot be retrieved later.
</Warning>

<ParamField body="name" type="string" required>
  A descriptive name for the token
</ParamField>

<ParamField body="type" type="string" required>
  Token type: `ingest` (for SDK telemetry) or `api` (for management/analytics)
</ParamField>

<ParamField body="permissions" type="array">
  Permissions for API tokens. Options:
  - `servers:read` - List and view server details
  - `servers:write` - Create, update, and delete servers
  - `analytics:read` - Query traces, metrics, and analytics
  - `tokens:manage` - Create and revoke other tokens
</ParamField>

<ResponseField name="success" type="boolean">
  Indicates if the request was successful
</ResponseField>

<ResponseField name="data" type="object">
  <Expandable title="properties">
    <ResponseField name="id" type="string">
      Token identifier
    </ResponseField>
    <ResponseField name="name" type="string">
      Token name
    </ResponseField>
    <ResponseField name="type" type="string">
      Token type
    </ResponseField>
    <ResponseField name="token" type="string">
      The full token value (only shown once)
    </ResponseField>
    <ResponseField name="permissions" type="array">
      Token permissions
    </ResponseField>
    <ResponseField name="created_at" type="string">
      Creation timestamp
    </ResponseField>
  </Expandable>
</ResponseField>

<RequestExample>
```bash cURL
curl -X POST https://api.app.shinzo.ai/v1/tokens \
  -H "Authorization: Bearer YOUR_API_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Production Server Token",
    "type": "api",
    "permissions": ["servers:read", "analytics:read"]
  }'
```

```python Python
import requests

response = requests.post(
    "https://api.app.shinzo.ai/v1/tokens",
    headers={
        "Authorization": "Bearer YOUR_API_TOKEN",
        "Content-Type": "application/json"
    },
    json={
        "name": "Production Server Token",
        "type": "api",
        "permissions": ["servers:read", "analytics:read"]
    }
)
print(response.json())
```

```typescript TypeScript
const response = await fetch("https://api.app.shinzo.ai/v1/tokens", {
```
