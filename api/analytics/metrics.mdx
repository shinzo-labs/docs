---
title: "Query Metrics"
api: "GET https://api.app.shinzo.ai/v1/analytics/metrics"
description: "Retrieve time-series metrics for visualization and analysis"
---

Query time-series metrics for building charts and dashboards. Supports various aggregation functions and time intervals.

<ParamField query="metric" type="string" required>
  Metric name. Options: `invocations`, `errors`, `error_rate`, `duration`, `throughput`
</ParamField>

<ParamField query="server_id" type="string">
  Filter by server identifier
</ParamField>

<ParamField query="tool_name" type="string">
  Filter by tool name
</ParamField>

<ParamField query="start_time" type="string">
  Start of time range (ISO 8601). Defaults to 24 hours ago.
</ParamField>

<ParamField query="end_time" type="string">
  End of time range (ISO 8601). Defaults to now.
</ParamField>

<ParamField query="interval" type="string" default="auto">
  Aggregation interval: `1m`, `5m`, `15m`, `1h`, `1d`, or `auto`
</ParamField>

<ParamField query="aggregation" type="string" default="avg">
  Aggregation function: `avg`, `sum`, `min`, `max`, `p50`, `p95`, `p99`
</ParamField>

<ResponseField name="success" type="boolean">
  Indicates if the request was successful
</ResponseField>

<ResponseField name="data" type="object">
  <Expandable title="properties">
    <ResponseField name="metric" type="string">
      The metric name
    </ResponseField>
    <ResponseField name="aggregation" type="string">
      The aggregation function used
    </ResponseField>
    <ResponseField name="interval" type="string">
      The time interval used
    </ResponseField>
    <ResponseField name="time_range" type="object">
      The queried time range
    </ResponseField>
    <ResponseField name="data_points" type="array">
      Array of timestamp/value pairs
    </ResponseField>
    <ResponseField name="summary" type="object">
      Summary statistics (min, max, avg, current)
    </ResponseField>
  </Expandable>
</ResponseField>

<RequestExample>
```bash cURL
curl -X GET "https://api.app.shinzo.ai/v1/analytics/metrics?metric=duration&aggregation=p95&interval=5m" \
  -H "Authorization: Bearer YOUR_API_TOKEN"
```

```python Python
import requests

response = requests.get(
    "https://api.app.shinzo.ai/v1/analytics/metrics",
    params={
        "metric": "duration",
        "aggregation": "p95",
        "interval": "5m"
    },
    headers={"Authorization": "Bearer YOUR_API_TOKEN"}
)
print(response.json())
```

```typescript TypeScript
const response = await fetch(
  "https://api.app.shinzo.ai/v1/analytics/metrics?metric=duration&aggregation=p95&interval=5m",
  {
    headers: { Authorization: "Bearer YOUR_API_TOKEN" }
  }
);
const data = await response.json();
```
