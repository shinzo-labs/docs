---
title: "Get Trace Details"
api: "GET https://api.app.shinzo.ai/v1/analytics/traces/{trace_id}"
description: "Retrieve detailed information about a specific trace"
---

Get the complete details of a trace including all spans, events, and attributes. Use this for deep debugging and performance analysis.

<ParamField path="trace_id" type="string" required>
  The unique trace identifier
</ParamField>

<ResponseField name="success" type="boolean">
  Indicates if the request was successful
</ResponseField>

<ResponseField name="data" type="object">
  <Expandable title="properties">
    <ResponseField name="trace_id" type="string">
      Unique trace identifier
    </ResponseField>
    <ResponseField name="server" type="object">
      Server information
    </ResponseField>
    <ResponseField name="session_id" type="string">
      Session identifier
    </ResponseField>
    <ResponseField name="spans" type="array">
      All spans in the trace with timing and attributes
    </ResponseField>
    <ResponseField name="events" type="array">
      Events that occurred during the trace
    </ResponseField>
  </Expandable>
</ResponseField>
